private var saldo: Double = 0.0

fun sacar(quantia: Double) {
if (quantia > saldo) {
println("Impossivel sacar. Saldo insuficiente.")
} else {
saldo -= quantia
println("Saque realizado com sucesso!")
}
}

fun depositar(quantia: Double) {
if (quantia < 0) {
println("Impossivel depositar valor negativo.")
} else {
saldo += quantia
println("Deposito realizado com sucesso!")
}
}

fun getSaldo(): Double {
return saldo
}
